  <script>
    // State management
    let currentConfig = JSON.parse(JSON.stringify(DEFAULT_RESIDENCE_CONFIG));
    let isPasswordVisible = true;
    let qrCodeInstance = null;

    // Initialize page on load
    document.addEventListener('DOMContentLoaded', () => {
      document.getElementById('current-year').textContent = new Date().getFullYear();
      
      // Load any locally saved host customizations
      try {
        const saved = localStorage.getItem('floria_residence_config');
        if (saved) {
          currentConfig = { ...currentConfig, ...JSON.parse(saved) };
        }
      } catch (e) {
        console.warn('Local storage unavailable, using default config.');
      }

      renderConfigToUI();
    });

    // Render current configuration values into DOM elements
    function renderConfigToUI() {
      document.getElementById('display-property-name').textContent = currentConfig.propertyName;
      document.getElementById('display-tagline').textContent = currentConfig.tagline;
      
      // Wi-Fi
      document.getElementById('display-ssid').textContent = currentConfig.wifi.ssid;
      document.getElementById('modal-ssid-label').textContent = currentConfig.wifi.ssid;
      updatePasswordDisplay();

      // Google Review
      document.getElementById('google-review-link').href = currentConfig.googleReviewUrl;

      // Social Links
      document.getElementById('social-instagram').href = currentConfig.socials.instagram;
      document.getElementById('social-tiktok').href = currentConfig.socials.tiktok;
      document.getElementById('social-facebook').href = currentConfig.socials.facebook;
      document.getElementById('social-whatsapp').href = currentConfig.socials.whatsapp;
    }

    // Toggle Wi-Fi password masking
    function togglePasswordVisibility() {
      isPasswordVisible = !isPasswordVisible;
      updatePasswordDisplay();
    }

    function updatePasswordDisplay() {
      const pwEl = document.getElementById('display-password');
      const eyeOpen = document.getElementById('eye-open-icon');
      const eyeClosed = document.getElementById('eye-closed-icon');

      if (isPasswordVisible) {
        pwEl.textContent = currentConfig.wifi.password;
        eyeOpen.classList.remove('hidden');
        eyeClosed.classList.add('hidden');
      } else {
        pwEl.textContent = '•'.repeat(Math.min(currentConfig.wifi.password.length, 14));
        eyeOpen.classList.add('hidden');
        eyeClosed.classList.remove('hidden');
      }
    }

    // Copy to clipboard with iframe-safe fallback & feedback
    function copyText(type) {
      const valueToCopy = type === 'ssid' ? currentConfig.wifi.ssid : currentConfig.wifi.password;
      const label = type === 'ssid' ? 'Network name copied!' : 'Wi-Fi password copied!';

      fallbackCopyTextToClipboard(valueToCopy, () => {
        showToast(label);

        if (type === 'password') {
          const btnLabel = document.getElementById('copy-btn-label');
          const btn = document.getElementById('copy-pw-btn');
          btnLabel.textContent = 'Copied!';
          btn.classList.add('bg-floria-gold', 'text-white');
          btn.classList.remove('bg-floria-charcoal');

          setTimeout(() => {
            btnLabel.textContent = 'Copy Password';
            btn.classList.remove('bg-floria-gold');
            btn.classList.add('bg-floria-charcoal');
          }, 2200);
        }
      });
    }

    function fallbackCopyTextToClipboard(text, onSuccess) {
      const textArea = document.createElement("textarea");
      textArea.value = text;
      textArea.style.position = "fixed";
      textArea.style.top = "-9999px";
      textArea.style.left = "-9999px";
      document.body.appendChild(textArea);
      textArea.focus();
      textArea.select();

      try {
        const successful = document.execCommand('copy');
        if (successful && onSuccess) onSuccess();
      } catch (err) {
        console.error('Fallback copy failed', err);
      }
      document.body.removeChild(textArea);
    }

    // Toast notification controller
    let toastTimeout;
    function showToast(message) {
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toast-message');
      toastMsg.textContent = message;

      toast.classList.remove('opacity-0', '-translate-y-2');
      toast.classList.add('opacity-100', 'translate-y-0');

      clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => {
        toast.classList.remove('opacity-100', 'translate-y-0');
        toast.classList.add('opacity-0', '-translate-y-2');
      }, 2500);
    }

    // =====================================================================
    // WI-FI QR CODE MODAL LOGIC
    // =====================================================================
    function escapeWifiField(str) {
      return String(str).replace(/([\\;,:"])/g, '\\$1');
    }

    function openWifiQrModal() {
      const modal = document.getElementById('wifi-qr-modal');
      const container = document.getElementById('qrcode-container');
      container.innerHTML = ''; // Clear previous QR

      // Standard Wi-Fi QR payload format: WIFI:T:WPA;S:MyNetwork;P:MyPassword;;
      const encryption = currentConfig.wifi.encryption || 'WPA';
      const ssid = escapeWifiField(currentConfig.wifi.ssid);
      const pass = escapeWifiField(currentConfig.wifi.password);
      const wifiPayload = `WIFI:T:${encryption};S:${ssid};P:${pass};;`;

      new QRCode(container, {
        text: wifiPayload,
        width: 190,
        height: 190,
        colorDark: "#1E1D1B",
        colorLight: "#FFFFFF",
        correctLevel: QRCode.CorrectLevel.M
      });

      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeWifiQrModal() {
      const modal = document.getElementById('wifi-qr-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    // =====================================================================
    // HOST CUSTOMIZATION DRAWER LOGIC
    // =====================================================================
    function openCustomizeModal() {
      document.getElementById('input-ssid').value = currentConfig.wifi.ssid;
      document.getElementById('input-password').value = currentConfig.wifi.password;
      document.getElementById('input-google-review').value = currentConfig.googleReviewUrl;
      document.getElementById('input-instagram').value = currentConfig.socials.instagram;
      document.getElementById('input-tiktok').value = currentConfig.socials.tiktok;
      document.getElementById('input-facebook').value = currentConfig.socials.facebook;
      document.getElementById('input-whatsapp').value = currentConfig.socials.whatsapp;

      const modal = document.getElementById('customize-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeCustomizeModal() {
      const modal = document.getElementById('customize-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function saveCustomDetails(event) {
      event.preventDefault();

      currentConfig.wifi.ssid = document.getElementById('input-ssid').value.trim();
      currentConfig.wifi.password = document.getElementById('input-password').value.trim();
      currentConfig.googleReviewUrl = document.getElementById('input-google-review').value.trim();
      currentConfig.socials.instagram = document.getElementById('input-instagram').value.trim();
      currentConfig.socials.tiktok = document.getElementById('input-tiktok').value.trim();
      currentConfig.socials.facebook = document.getElementById('input-facebook').value.trim();
      currentConfig.socials.whatsapp = document.getElementById('input-whatsapp').value.trim();

      try {
        localStorage.setItem('floria_residence_config', JSON.stringify(currentConfig));
      } catch (e) {}

      renderConfigToUI();
      closeCustomizeModal();
      showToast('Details updated!');
    }

    function copyConfigSnippet() {
      const updatedObj = {
        propertyName: currentConfig.propertyName,
        tagline: currentConfig.tagline,
        wifi: {
          ssid: document.getElementById('input-ssid').value.trim(),
          password: document.getElementById('input-password').value.trim(),
          encryption: "WPA"
        },
        googleReviewUrl: document.getElementById('input-google-review').value.trim(),
        socials: {
          instagram: document.getElementById('input-instagram').value.trim(),
          tiktok: document.getElementById('input-tiktok').value.trim(),
          facebook: document.getElementById('input-facebook').value.trim(),
          whatsapp: document.getElementById('input-whatsapp').value.trim()
        }
      };

      const snippet = `const DEFAULT_RESIDENCE_CONFIG = ${JSON.stringify(updatedObj, null, 2)};`;
      fallbackCopyTextToClipboard(snippet, () => {
        showToast('Config code copied to clipboard!');
      });
    }

    function resetToDefaults() {
      try {
        localStorage.removeItem('floria_residence_config');
      } catch (e) {}
      currentConfig = JSON.parse(JSON.stringify(DEFAULT_RESIDENCE_CONFIG));
      renderConfigToUI();
      closeCustomizeModal();
      showToast('Reset to default values');
    }

    // Close modals when clicking outside content box
    window.addEventListener('click', (e) => {
      const wifiModal = document.getElementById('wifi-qr-modal');
      const customModal = document.getElementById('customize-modal');
      if (e.target === wifiModal) closeWifiQrModal();
      if (e.target === customModal) closeCustomizeModal();
    });
